04 — Reutilización

Cómo el servicio y el partial view son compartidos entre Termino, Articulo y PerfilMedico.

1. IValidacionContenidoService — interfaz compartida

Archivo: Services/Validacion/IValidacionContenidoService.cs

public interface IValidacionContenidoService
{
    // Upsert: inserta si no existe, actualiza si ya existe
    // Retorna true = creado, false = actualizado
    Task<bool> UpsertAsync(
        TipoContenidoValidado tipo,
        int contenidoId,
        string usuarioMedicoId,
        string? comentario);

    // Devuelve el registro si existe y es visible (Estado == Validado)
    Task<ValidacionContenidoProfesional?> ObtenerValidadoAsync(
        TipoContenidoValidado tipo,
        int contenidoId);

    // Devuelve el registro del médico (cualquier estado) — para el panel manage
    Task<ValidacionContenidoProfesional?> ObtenerDelMedicoAsync(
        TipoContenidoValidado tipo,
        int contenidoId,
        string usuarioMedicoId);

    // Admin: cambiar estado
    Task CambiarEstadoAsync(
        int validacionId,
        EstadoValidacion nuevoEstado,
        string adminId,
        string? nota);
}

Esta interfaz es la misma para los tres tipos de contenido. El parámetro tipo diferencia el contexto. No se duplica lógica.

2. ValidacionContenidoService — implementación única

Archivo: Services/Validacion/ValidacionContenidoService.cs

public class ValidacionContenidoService : IValidacionContenidoService
{
    private readonly ApplicationDbContext _context;

    public ValidacionContenidoService(ApplicationDbContext context)
        => _context = context;

    public async Task<bool> UpsertAsync(
        TipoContenidoValidado tipo, int contenidoId,
        string usuarioMedicoId, string? comentario)
    {
        var existente = await _context.ValidacionesContenidoProfesional
            .FirstOrDefaultAsync(v =>
                v.TipoContenido == tipo
                && v.ContenidoId == contenidoId
                && v.UsuarioMedicoId == usuarioMedicoId);

        if (existente is null)
        {
            _context.ValidacionesContenidoProfesional.Add(new ValidacionContenidoProfesional
            {
                TipoContenido  = tipo,
                ContenidoId    = contenidoId,
                UsuarioMedicoId = usuarioMedicoId,
                Comentario     = comentario,
                Estado         = EstadoValidacion.Validado,
                CreadoEn       = DateTime.UtcNow,
                ActualizadoEn  = DateTime.UtcNow
            });
            await _context.SaveChangesAsync();
            return true;   // creado
        }

        existente.Comentario    = comentario;
        existente.ActualizadoEn = DateTime.UtcNow;
        // NOTA: NO se resetea el Estado — si admin lo puso EnRevision, se respeta
        await _context.SaveChangesAsync();
        return false;  // actualizado
    }

    // ... ObtenerValidadoAsync, ObtenerDelMedicoAsync, CambiarEstadoAsync ...
}

4. Partial view compartido: _ValidacionContenidoMedico.cshtml

Archivo: Pages/Shared/_ValidacionContenidoMedico.cshtml

@model ValidacionContenidoPartialVm

@if (Model.ValidacionExistente is null)
{
    <!-- Primera validación -->
    <form method="post" asp-page-handler="ValidarContenido">
        <input type="hidden" name="tipoContenido" value="@((int)Model.TipoContenido)" />
        <input type="hidden" name="contenidoId"   value="@Model.ContenidoId" />
        <textarea name="comentario" class="form-control mb-2"
                  placeholder="Nota interna (opcional)" maxlength="800"></textarea>
        <button type="submit" class="btn btn-success btn-sm">
            ✓ Validar mi perfil
        </button>
    </form>
}
else if (Model.ValidacionExistente.Estado == EstadoValidacion.Validado)
{
    <!-- Ya validado — mostrar estado y opción de actualizar -->
    <div class="alert alert-success py-2">
        <strong>✓ Validado</strong> — @Model.ValidacionExistente.ActualizadoEn.ToLocalTime().ToString("dd/MM/yyyy")
    </div>
    <form method="post" asp-page-handler="ValidarContenido">
        <input type="hidden" name="tipoContenido" value="@((int)Model.TipoContenido)" />
        <input type="hidden" name="contenidoId"   value="@Model.ContenidoId" />
        <textarea name="comentario" class="form-control mb-2" maxlength="800"
            >@Model.ValidacionExistente.Comentario</textarea>
        <button type="submit" class="btn btn-outline-success btn-sm">Actualizar validación</button>
    </form>
}
else
{
    <div class="alert alert-warning py-2">
        Validación en revisión por el equipo. Contacta a soporte si tienes dudas.
    </div>
}
