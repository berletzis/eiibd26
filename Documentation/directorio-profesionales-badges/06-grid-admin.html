Función JS revocarBadgeConConfirmacion

async function revocarBadgeConConfirmacion(medicoId, codigo, nombre) {
    if (!confirm(`¿Revocar badge "${nombre}" de este profesional?\n\nEsta acción quedará registrada en el historial.`)) return;
    try {
        const resp = await fetch('?handler=RevocarBadge', {
            method: 'POST',
            headers: { 'Content-Type': 'application/x-www-form-urlencoded', 'RequestVerificationToken': getAntiForgeryToken() },
            body: `medicoId=${medicoId}&codigo=${encodeURIComponent(codigo)}`
        });
        const result = await resp.json();
        if (result.success) {
            showToast('Badge revocado correctamente.', 'success');
            loadMedicoPanel(medicoId);  // refrescar el panel
        } else {
            showToast('No se pudo revocar el badge.', 'danger');
        }
    } catch {
        showToast('Error de red al revocar badge.', 'danger');
    }
}

// Función JS:
async function recalcularNivel(medicoId) {
    const resp = await fetch('?handler=RecalcularNivel', {
        method: 'POST',
        headers: { 'Content-Type': 'application/x-www-form-urlencoded', 'RequestVerificationToken': getAntiForgeryToken() },
        body: `medicoId=${medicoId}`
    });
    const result = await resp.json();
    if (result.success) {
        showToast(`Nivel recalculado: ${result.nuevoNivel}`, 'success');
        loadMedicoPanel(medicoId);
    }
}

public async Task<IActionResult> OnPostRecalcularNivelAsync(int medicoId)
{
    await _service.RecalcularNivelConfianzaAsync(medicoId);
    var m = await _db.MedicosDirectorio.AsNoTracking().FirstOrDefaultAsync(x => x.Id == medicoId);
    return new JsonResult(new { success = true, nuevoNivel = m?.NivelConfianza.ToString() ?? "—" });
}
