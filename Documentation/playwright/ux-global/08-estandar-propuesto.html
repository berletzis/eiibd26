  --text-md:       18px;   /* subtítulos grandes */
  --text-lg:       22px;   /* H2 de sección */
  --text-xl:       28px;   /* H1 de página */

  /* Pesos */
  --fw-normal:     400;
  --fw-medium:     500;
  --fw-semibold:   600;
  --fw-bold:       700;

  /* Interlineado */
  --lh-tight:      1.2;   /* títulos */
  --lh-base:       1.5;   /* cuerpo */
  --lh-loose:      1.75;  /* texto largo / artículos */
}

/* Aplicación al body */
