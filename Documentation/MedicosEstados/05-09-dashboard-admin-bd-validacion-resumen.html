public class PropuestaPacienteVm
{
    public int    Id          { get; set; }
    public string Nombre      { get; set; } = string.Empty;
    public string? Especialidad { get; set; }
    public string? Estado       { get; set; }
    public DateTimeOffset Fecha { get; set; }
    public EstadoProfesionalDerivado EstadoDerivado { get; set; }

    public string EstadoLabel => EstadoDerivado switch
    {
        EstadoProfesionalDerivado.Propuesto        => "En revisión",
        EstadoProfesionalDerivado.CedulaVerificada => "Cédula verificada",
        EstadoProfesionalDerivado.Publicado        => "Publicado en el directorio",
        EstadoProfesionalDerivado.Reclamado        => "Reclamado por el profesional",
        _                                          => "En revisión"
    };

    public string EstadoBadgeClass => EstadoDerivado switch
    {
        EstadoProfesionalDerivado.Propuesto        => "bg-warning text-dark",
        EstadoProfesionalDerivado.CedulaVerificada => "bg-success",
        EstadoProfesionalDerivado.Publicado        => "bg-primary",
        EstadoProfesionalDerivado.Reclamado        => "bg-info text-dark",
        _                                          => "bg-secondary"
    };
}

Query corregida en el servicio

// En MedicoDirectorioService.cs
public async Task<List<PropuestaPacienteVm>> GetMisPropuestasAsync(Guid usuarioId)
{
    return await _db.MedicosDirectorio
        .AsNoTracking()
        .IgnoreQueryFilters()                       // ← incluir eliminados para no mostrarlos
        .Where(m => m.PropuestoPorUsuarioId == usuarioId
                 && !m.Eliminado)                   // ← excluir eliminados del dashboard
        .OrderByDescending(m => m.FechaCreacion)
        .Select(m => new PropuestaPacienteVm
        {
            Id          = m.Id,
            Nombre      = m.NombreCompleto,
            Especialidad = m.Especialidad,
            Estado       = m.Estado,
            Fecha        = m.FechaCreacion,
            // Calcular EstadoDerivado en C# (no en SQL)
            EstadoDerivado = m.Eliminado ? EstadoProfesionalDerivado.Eliminado
                           : m.Activo && m.VisiblePublicamente && m.EstatusReclamacion == EstatusReclamacion.Reclamado
                             ? EstadoProfesionalDerivado.Reclamado
                           : m.Activo && m.VisiblePublicamente
                             ? EstadoProfesionalDerivado.Publicado
                           : m.EstatusValidacion == EstatusValidacionCedula.Validado
                             ? EstadoProfesionalDerivado.CedulaVerificada
                           : EstadoProfesionalDerivado.Propuesto
        })
        .ToListAsync();
}

El panel lateral del admin mezcla datos personales, verificación, badges y auditoría sin jerarquía clara. El nuevo orden agrupa información relacionada:

Datos profesionales — Nombre, Especialidad, Subespecialidad, Cédula, País

// Helper JS para estado derivado:
function estadoLabel(d) {
    if (d.eliminado)      return '🗑 Eliminado';
    if (d.perfilReclamado)    return '🔐 Reclamado';
    if (d.activo && d.visible) return '🌐 Publicado';
    if (d.cedulaVerificada)   return '✅ Cédula verificada';
    return '📋 Propuesto / En revisión';
}
function estadoBadge(d) {
    if (d.eliminado)          return 'bg-secondary';
    if (d.perfilReclamado)    return 'bg-success';
    if (d.activo && d.visible) return 'bg-primary';
    if (d.cedulaVerificada)   return 'bg-info text-dark';
    return 'bg-warning text-dark';
}

-- 1. Cuántos profesionales por estado derivado
SELECT
    SUM(CASE WHEN Eliminado = 1 THEN 1 ELSE 0 END) AS Eliminados,
    SUM(CASE WHEN Eliminado = 0 AND Activo = 1 AND VisiblePublicamente = 1
             AND EstatusReclamacion = 2 THEN 1 ELSE 0 END) AS Reclamados,
    SUM(CASE WHEN Eliminado = 0 AND Activo = 1 AND VisiblePublicamente = 1
             AND EstatusReclamacion != 2 THEN 1 ELSE 0 END) AS Publicados,
    SUM(CASE WHEN Eliminado = 0 AND (Activo = 0 OR VisiblePublicamente = 0)
             AND EstatusValidacion = 1 THEN 1 ELSE 0 END) AS CedulaVerificados,
    SUM(CASE WHEN Eliminado = 0 AND (Activo = 0 OR VisiblePublicamente = 0)
             AND EstatusValidacion != 1 THEN 1 ELSE 0 END) AS Propuestos
FROM MedicosDirectorio;

-- 2. Inconsistencias badge "verificado" vs EstatusValidacion
SELECT md.Id, md.NombreCompleto, md.EstatusValidacion,
    (SELECT COUNT(*) FROM MedicoPerfilBadge mpb
     JOIN MedicoBadge mb ON mpb.BadgeId = mb.Id
     WHERE mpb.MedicoId = md.Id AND mb.Codigo = 'verificado') AS TieneBadgeVerificado
FROM MedicosDirectorio md
WHERE md.Eliminado = 0
  AND (
    -- Tiene cédula verificada pero no tiene badge
    (md.EstatusValidacion = 1 AND NOT EXISTS (
        SELECT 1 FROM MedicoPerfilBadge mpb
        JOIN MedicoBadge mb ON mpb.BadgeId = mb.Id
        WHERE mpb.MedicoId = md.Id AND mb.Codigo = 'verificado'))
    OR
    -- Tiene badge pero no tiene cédula verificada
    (md.EstatusValidacion != 1 AND EXISTS (
        SELECT 1 FROM MedicoPerfilBadge mpb
        JOIN MedicoBadge mb ON mpb.BadgeId = mb.Id
        WHERE mpb.MedicoId = md.Id AND mb.Codigo = 'verificado'))
  );

DECLARE @BadgeId INT = (SELECT Id FROM MedicoBadge WHERE Codigo = 'verificado');

-- Otorgar badge "verificado" a profesionales con cédula verificada que no lo tienen
INSERT INTO MedicoPerfilBadge (MedicoId, BadgeId, FechaObtenido, OtorgadoPor)
SELECT md.Id, @BadgeId, GETUTCDATE(), 'sistema-repair'
FROM MedicosDirectorio md
WHERE md.EstatusValidacion = 1   -- Validado
  AND md.Eliminado = 0
  AND NOT EXISTS (
      SELECT 1 FROM MedicoPerfilBadge mpb
      WHERE mpb.MedicoId = md.Id AND mpb.BadgeId = @BadgeId
  );

-- Revocar badge "verificado" de profesionales sin cédula verificada
DELETE FROM MedicoPerfilBadge
WHERE BadgeId = @BadgeId
  AND MedicoId IN (
      SELECT md.Id FROM MedicosDirectorio md
      WHERE md.EstatusValidacion != 1  -- No validado
  );

FASE 8 — Validación: Escenario Completo

Caso de prueba E2E — flujo completo

Escenario 1: Paciente propone → Admin verifica → Paciente ve estado actualizado

GIVEN paciente autenticado en /DirectorioMedicos
WHEN propone "Dr. Alejandro Reyes Ortiz"
THEN aparece en "Mis aportaciones" con badge "En revisión"
WHEN admin abre admin grid, abre panel lateral del profesional, marca "Cédula verificada"
THEN en el panel: sección Verificación muestra ✅, badge "verificado" aparece en sección Badges, EstadoDerivado dice "Cédula verificada"
WHEN paciente refresca /DirectorioMedicos
THEN badge del profesional dice "Cédula verificada" (NO "En revisión")

Escenario 2: Soft delete y restauración

GIVEN admin grid con profesionales activos
WHEN admin hace clic en eliminar sobre un registro
THEN el registro desaparece del grid principal
WHEN admin activa switch "Eliminados"
THEN el registro eliminado APARECE en el grid con badge "Eliminado"
THEN solo aparece el botón "Restaurar" (no "Eliminar") para ese registro
WHEN admin hace clic en "Restaurar"
THEN el registro desaparece del grid de eliminados
WHEN admin desactiva el switch "Eliminados"
THEN el registro restaurado REAPARECE en el grid principal

Escenario 3: Consistencia de badges

GIVEN profesional con EstatusValidacion=Pendiente y sin badge "verificado"
WHEN admin abre panel → sección Badges muestra "verificado = Pendiente"
WHEN admin marca checkbox "Cédula verificada" y guarda
THEN panel se recarga → sección Badges muestra "verificado = Obtenido"
THEN sección superior muestra ✅ Cédula verificada
THEN ambas secciones son CONSISTENTES
WHEN admin desmarca checkbox y guarda
THEN badge "verificado" vuelve a "Pendiente" en ambas secciones
